\textbf{Consignes.} Travail individuel, sans Internet. Documents de cours autorisés.
\begin{itemize}
  \item Créer un nouveau classeur LibreOffice Calc et l'\textbf{enregistrer immédiatement} (\menu{Fichier > Enregistrer sous})
        sous le nom \texttt{NOM\_Prenom\_Eval\_Tableur.ods}. Enregistrer régulièrement.
  \item \textbf{Toute valeur calculée doit l'être par une formule} : un nombre recopié à la main vaut 0.
        En cas d'erreur de saisie des données, les calculs sont notés sur la justesse des formules.
  \item Les réponses de la partie D se rédigent sur cette feuille, à rendre avec le fichier.
\end{itemize}
\end{cdc}

\textbf{Contexte.} L'entreprise \emph{NetCIEL} vend du matériel réseau. Le responsable commercial vous confie
ses ventes 2025 (quantités vendues par trimestre) pour établir un bilan.

\begin{partie}{Partie 0 — Saisie des données}{1 pt}

%---------------------------------------------------------------------
\begin{partie}{Partie A — Calculs avec formules et recopie}{9 pts}
\begin{enumerate}[label=\textbf{A\arabic*.}]
  \item En \cel{G4}, calculer le nombre total d'articles vendus dans l'année avec la fonction \cel{SOMME},
        puis recopier la formule jusqu'en \cel{G7} (poignée de recopie). \pts{2 pts}
  \item En \cel{H4}, calculer la moyenne trimestrielle des ventes avec la fonction \cel{MOYENNE}, puis recopier
        jusqu'en \cel{H7}. \pts{1,5 pt}
  \item En \cel{I4}, calculer le chiffre d'affaires annuel du produit (prix unitaire $\times$ total des quantités),
        puis recopier jusqu'en \cel{I7}. \pts{2 pts}
  \item Ligne 8 : calculer en \cel{C8} le total des ventes du 1\textsuperscript{er} trimestre, puis recopier
        la formule vers la droite jusqu'en \cel{G8}. Calculer aussi le CA total en \cel{I8}. \pts{1,5 pt}
  \item En \cel{J4}, calculer la part du CA du produit dans le CA total (\cel{I4} divisé par le CA total).
        La formule doit pouvoir être recopiée jusqu'en \cel{J7} \textbf{sans être modifiée} :
        utiliser une \textbf{référence absolue} (symbole \cel{\$}). \pts{2 pts}
\end{enumerate}
\end{partie}

%---------------------------------------------------------------------
\begin{partie}{Partie B — Mise en forme}{3 pts}
\begin{enumerate}[label=\textbf{B\arabic*.}]
  \item Fusionner les cellules \cel{A1:J1} ; titre centré, en gras, taille 14. \pts{1 pt}
  \item En-têtes de la ligne 3 en gras ; bordures sur tout le tableau \cel{A3:J8}. \pts{1 pt}
  \item Format \textbf{monétaire en euros} pour les colonnes \cel{B} et \cel{I} ;
        format \textbf{pourcentage avec 1 décimale} pour la colonne \cel{J}. \pts{1 pt}
\end{enumerate}
\end{partie}

\newpage
%---------------------------------------------------------------------
\begin{partie}{Partie C — Tri et graphique}{4 pts}
\begin{enumerate}[label=\textbf{C\arabic*.}]
  \item Trier les produits (lignes 4 à 7 seulement, \textbf{sans la ligne Total}) par CA annuel
        \textbf{décroissant} (\menu{Données > Trier}). Vérifier que les résultats sont restés justes. \pts{1,5 pt}
  \item Insérer un graphique en \textbf{secteurs} représentant la répartition du CA annuel par produit
        (noms des produits et CA ; touche \textbf{Ctrl} pour sélectionner deux plages non adjacentes)
        (\menu{Insertion > Diagramme}), avec le titre « Répartition du CA 2025 ». Le placer à côté du tableau,
        sans masquer de données. \pts{2,5 pts}
\end{enumerate}
\end{partie}

%---------------------------------------------------------------------
\begin{partie}{Partie D — Analyse (à rédiger sur cette feuille)}{3 pts}
\begin{enumerate}[label=\textbf{D\arabic*.}]
  \item Écrire la formule saisie en \cel{J4}. Que se serait-il passé en \cel{J5} si l'on avait recopié la formule
        \cel{=I4/I8}, sans le symbole \cel{\$} ? Expliquer. \pts{1,5 pt}
\zone{2.8cm}{\cel{=I4/\$I\$8}. Sans \cel{\$}, la recopie décale les deux références : \cel{J5} contiendrait
\cel{=I5/I9}. \cel{I9} est vide, d'où l'erreur \cel{\#DIV/0!}. Le \cel{\$} bloque la colonne et la ligne de la cellule du total.}
  \item Le produit le plus vendu en quantité est-il celui qui rapporte le plus ? Justifier avec vos résultats. \pts{1,5 pt}
\zone{2.8cm}{\ifsujetB Non. Le cordon fibre est le plus vendu (240 lots) mais ne rapporte que 7\,200~€ (24,2~\%) à cause
de son faible prix (30~€). Le switch PoE rapporte le plus (8\,400~€, 28,3~\%) avec 40 unités seulement.
\else Non. Le câble RJ45 est le plus vendu (193 lots) mais rapporte le moins (6\,755~€, 20,9~\%) à cause de son
faible prix (35~€). Le switch rapporte le plus (9\,000~€, 27,8~\%) avec 50 unités seulement.\fi}
\end{enumerate}
\end{partie}

\textbf{Bonus (+1 pt)} — En \cel{K4}, afficher « Best-seller » si le total des quantités est supérieur ou égal
à 50, et rien sinon (fonction \cel{SI}), puis recopier jusqu'en \cel{K7}.

\textbf{En fin d'épreuve} : enregistrer, fermer le fichier, le déposer dans le dossier indiqué, rendre cette feuille.
