\section{Contexte}
\activite{} \textbf{\Entreprise} possède un \textbf{\sC} et deux sites (\textbf{\sUn} et \textbf{\sDeux}). Elle vous confie
la conception complète de son réseau sous Cisco Packet Tracer. Comme dans le TP de synthèse, \textbf{aucune procédure pas
à pas n'est fournie} : vous mobilisez en autonomie les TP2.1 (adressage IP), TP2.2 (concentrateur et commutateur) et TP2.3
(routeur, passerelle), et vous faites les bons choix techniques. Nouveauté : les sites sont reliés par \textbf{trois routeurs}.

\textbf{Compétences évaluées :} concevoir un plan d'adressage cohérent ; choisir et câbler les équipements ;
configurer postes, passerelles et interfaces de routeurs ; établir le \textbf{routage statique} entre trois routeurs ;
valider la connectivité et justifier ses choix.

\textbf{Contraintes générales :}
\begin{itemize}
  \item Trois routeurs \textbf{Cisco 2911} ; chaque réseau constitue un \textbf{sous-réseau distinct}.
  \item Passerelle de chaque réseau local = \textbf{dernière adresse utilisable}. Sur chaque liaison inter-routeurs,
        \rC{} prend l'adresse \textbf{.1} et le routeur de site l'adresse \textbf{.2}.
  \item Routage \textbf{statique uniquement} (routes statiques et/ou route par défaut ; pas de RIP).
  \item Chaque hôte doit pouvoir « \texttt{ping} » \textbf{n'importe quel autre hôte} et afficher la page web du serveur.
  \item Choix concentrateur/commutateur \textbf{justifié} ; équipements \textbf{nommés clairement} (ex. \ip{PC-\pUn1}, \ip{SW-\pUn}).
\end{itemize}

\begin{aide}\textbf{Aide Packet Tracer — route statique.} Routeur $\to$ onglet \textbf{Config} $\to$ \textbf{ROUTING / Static} :
\emph{Network} = adresse du réseau à atteindre, \emph{Mask} = son masque, \emph{Next Hop} = adresse IP de l'interface du
\textbf{routeur voisin} sur la liaison, puis \textbf{Add}. Une \textbf{route par défaut} s'écrit \emph{Network}
\texttt{0.0.0.0}, \emph{Mask} \texttt{0.0.0.0}. Un routeur connaît d'office ses réseaux directement connectés.\end{aide}

\section{Questions de réflexion \hfill /4}

\question{1}{1 pt}{Avez-vous choisi des concentrateurs ou des commutateurs pour les deux sites ? Justifiez
(rôle, domaine de collision, performances).}
\zone{2.3cm}{\textbf{Commutateurs.} Un hub répète chaque trame sur tous ses ports (un seul domaine de collision,
bande passante partagée). Le switch utilise sa \textbf{table MAC} pour n'envoyer la trame que sur le port du
destinataire : un domaine de collision par port, moins de trafic inutile, meilleures performances (et confidentialité).}

\question{2}{1 pt}{Quel câble (droit ou croisé) avez-vous utilisé pour chaque liaison : poste–commutateur,
commutateur–routeur, routeur–routeur, serveur–routeur ?}
\zone{1.3cm}{Poste–commutateur : \textbf{droit} ; commutateur–routeur : \textbf{droit} ; routeur–routeur : \textbf{croisé} ;
serveur–routeur : \textbf{croisé} (équipements de même type). \emph{0,25 pt par liaison.}}

\question{3}{1 pt}{Adresses et passerelles configurées, mais routes pas encore ajoutées : PC-\pUn1 ne joint pas \srvNom.
Pourquoi ? Qu'apporte une route statique ?}
\zone{2.3cm}{PC-\pUn1 envoie bien le paquet à sa passerelle \rUn, mais \rUn{} ne connaît que ses réseaux
\textbf{directement connectés} (\netUn.0/24 et \lkUn.0/24) : il ignore le réseau \netSrv.0/24 et rejette le paquet
(« Destination host unreachable »). Une \textbf{route statique} indique au routeur, pour un réseau de destination,
l'adresse du \textbf{routeur suivant} (next hop) à qui transmettre le paquet.}

\question{4}{0,5 pt}{Pour l'adresse \ip{\qIP} : classe, adresse réseau, adresse de diffusion et nombre d'hôtes adressables.}
\zone{1.3cm}{Classe \textbf{A}, masque 255.0.0.0 ; réseau \textbf{10.0.0.0} ; diffusion \textbf{10.255.255.255} ;
$2^{24}-2 =$ \textbf{16\,777\,214} hôtes.}

\question{5}{0,5 pt}{On supprime sur \rC{} la route vers le réseau \lanUn. Le ping PC-\pUn1 $\to$ \srvNom{} échoue alors
que la requête atteint bien le serveur. Expliquez.}
\zone{1.8cm}{La requête passe (route par défaut de \rUn{} puis réseau connecté de \rC), le serveur répond, mais \rC{}
ne sait plus où envoyer la réponse destinée à \netUn.0/24 : elle est perdue. Le routage doit être correct \textbf{dans
les deux sens} (aller et retour).}
